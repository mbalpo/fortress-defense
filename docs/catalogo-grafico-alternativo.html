Tutto quello che si vede qui è disegnato dal codice del gioco: il modulo app/draw-hd2d.js e lo script di app/index.html, incorporati in questa pagina. Ogni schermata è il gioco vero, fermato in uno stato preciso e mosso da un orologio che si controlla dalla barra (play/pausa, un fotogramma alla volta, rallentatore). Le tessere piccole usano le stesse funzioni (TURRET_ART, drawEnemy, statusIcon, FX…) con valori di esempio.

Vincoli per chi propone modifiche

  • Canvas 2D, logico 360×640 (si scala allo schermo). Nessuno sprite e nessuna texture: solo forme semplici (rettangoli, cerchi, poligoni, linee, gradienti). Fanno eccezione solo le icone della PWA.
  • Leggibilità a 360×640 su telefono: nemici, proiettili e indicatori devono capirsi a colpo d’occhio, anche con 4 torrette e molti nemici insieme (§ 6 di questa pagina).
  • Il colore indica il tipo di danno (Force, Energy, Electric, Fire, Physical): torrette, loro effetti e nemici usano il colore del proprio tipo, perché serve per leggere resistenze e debolezze. Vedi la palette (§ 7).
  • Grafica e testi originali: il gioco è ispirato a un tower defense esistente ma non ne copia nomi, personaggi, disegni o aspetto.
  • Nel gioco il personaggio si chiama “guardiano”. Testi in italiano, apostrofo tipografico ’ e virgolette “ ”.
  • Le modifiche si propongono come funzioni di app/draw.js con la stessa firma (stessi argomenti, nessun valore di ritorno nuovo); qui solo disegno, nessuna regola di gioco. Per un proiettile nuovo: una voce in FX. Prestazioni: tutto si ridisegna 60 volte al secondo.

function mkCanvas(w, h) { const c = document.createElement("canvas"); c.width = Math.max(1, Math.ceil(w)); c.height = Math.max(1, Math.ceil(h)); return c; }
function hexRgb(h) { const n = parseInt(h.slice(1), 16); return h.length === 4 ? [((n >> 8) & 15) * 17, ((n >> 4) & 15) * 17, (n & 15) * 17] : [(n >> 16) & 255, (n >> 8) & 255, n & 255]; }
function mix(a, b, t) { const A = hexRgb(a), B = hexRgb(b); return "#" + A.map((v, i) => Math.round(v + (B[i] - v) * t).toString(16).padStart(2, "0")).join(""); }
function rgba(h, a) { const c = hexRgb(h); return "rgba(" + c[0] + "," + c[1] + "," + c[2] + "," + a + ")"; }

// funzione → canvas, con il contorno scuro aggiunto da solo attorno ai pixel pieni

  return procSprite(key, size, size, (i, j) => {
    const dx = i - R, dy = j - R, d = Math.hypot(dx, dy), a = Math.atan2(dy, dx) - rot;
    const u = ((a / 6.2832) % 1 + 1) % 1 * n, i0 = Math.floor(u) % n, f = u - Math.floor(u);
    const rr = (S[i0] * (1 - f) + S[(i0 + 1) % n] * f) * R * (1 + 0.09 * Math.sin(a * 5 + e.__hdId) + 0.06 * Math.sin(a * 11 + e.__hdId * 2));   // bitorzoli
    if (d > rr + 0.3) return null;

    let k = lam > 0.78 ? 5 : lam > 0.52 ? 4 : lam > 0.22 ? 3 : lam > -0.08 ? 2 : 1;
    for (const [ca, cd, cr] of craters) {                    // crateri che girano con la roccia
      const cx = Math.cos(ca + rot) * cd * R, cy = Math.sin(ca + rot) * cd * R, q = Math.hypot(dx - cx, dy - cy) / (cr * R);
      if (q < 0.75) k = Math.max(1, k - 2); else if (q < 1.05 && (dx - cx) + (dy - cy) > 0) k = Math.min(5, k + 1);

  });
}

  const inPoly = (pts, x, y) => { let c = false; for (let a = 0, b = pts.length - 1; a < pts.length; b = a++) { const [xa, ya] = pts[a], [xb, yb] = pts[b]; if ((ya > y) !== (yb > y) && x < (xb - xa) * (y - ya) / (yb - ya) + xa) c = !c; } return c; };

  });
